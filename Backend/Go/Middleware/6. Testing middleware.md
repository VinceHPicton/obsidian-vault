func TestAuthMiddleware(t *testing.T) {
	next := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
	})
	request := httptest.NewRequest(http.MethodGet, "/users", nil)
	recorder := httptest.NewRecorder()

	tokenMgr, err := tokens.New("super-secret-code")
	if err != nil {
		t.Fatal(err)
	}

	AuthMiddleware(tokenMgr)(next).ServeHTTP(recorder, request)

	if recorder.Code != http.StatusUnauthorized {
		t.Errorf("expected 401, got %d", recorder.Code)
	}
}

If your `Middleware` is hard to construct outside a `Server`, that's a sign it has too many hidden dependencies — good middleware constructors take exactly what they need (a token validator, a logger) rather than the whole `*Server`.
